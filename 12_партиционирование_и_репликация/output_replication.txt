=================== 1. primary: публикация таблиц titles и checkout_facts ===================
CREATE PUBLICATION pub_lib FOR TABLE titles, checkout_facts
CREATE PUBLICATION

=================== 2. replica: схема переносится вручную (DDL не реплицируется), затем подписка ===================
CREATE DATABASE
NOTICE:  created replication slot "sub_lib" on publisher
CREATE SUBSCRIPTION sub_lib CONNECTION 'host=hpd-w12-primary dbname=seattle_library user=postgres password=***' PUBLICATION pub_lib
CREATE SUBSCRIPTION
-- начальная синхронизация (COPY) завершена
SELECT (SELECT COUNT(*) FROM titles) AS titles, (SELECT COUNT(*) FROM checkout_facts) AS facts
