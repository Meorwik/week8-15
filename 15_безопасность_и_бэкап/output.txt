===== Часть 3. Резервная копия =====
--- 1. число строк до копии ---
$ psql -U postgres -d seattle_library -c SELECT 'titles' AS tbl, COUNT(*) FROM titles UNION ALL SELECT 'checkout_facts', COUNT(*) FROM checkout_facts UNION ALL SELECT 'subjects', COUNT(*) FROM subjects UNION ALL SELECT 'title_subjects', COUNT(*) FROM title_subjects;
      tbl       | count 
----------------+-------
 titles         |  5000
 checkout_facts |  5000
 subjects       |  7298
 title_subjects | 17988
(4 rows)

--- 2. копия в формате custom (-Fc) ---
$ pg_dump -U postgres -Fc -f /tmp/seattle_library.dump seattle_library
$ ls -l /tmp/seattle_library.dump
-rw-r--r-- 1 root root 1098555 Oct 22 05:23 /tmp/seattle_library.dump
--- 3. «авария»: удалили таблицу ---
$ psql -U postgres -d seattle_library -c DROP TABLE title_subjects;
DROP TABLE
$ psql -U postgres -d seattle_library -c SELECT COUNT(*) FROM title_subjects;
ERROR:  relation "title_subjects" does not exist
LINE 1: SELECT COUNT(*) FROM title_subjects;
                             ^
--- 4. восстановление в новую базу ---
$ createdb -U postgres seattle_restore
$ pg_restore -U postgres -d seattle_restore /tmp/seattle_library.dump
(pg_restore завершился без ошибок)
--- 5. число строк в восстановленной базе ---
$ psql -U postgres -d seattle_restore -c SELECT 'titles' AS tbl, COUNT(*) FROM titles UNION ALL SELECT 'checkout_facts', COUNT(*) FROM checkout_facts UNION ALL SELECT 'subjects', COUNT(*) FROM subjects UNION ALL SELECT 'title_subjects', COUNT(*) FROM title_subjects;
      tbl       | count 
----------------+-------
 titles         |  5000
 checkout_facts |  5000
 subjects       |  7298
 title_subjects | 17988
(4 rows)
